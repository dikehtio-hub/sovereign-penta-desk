    def _authenticate_tradovate(self):        if time.time() < self.token_expiry and self.tradovate_token:            return self.tradovate_token

        return False
    def validate_pre_trade_risk(self, phase: str, symbol: str, payload_passphrase: str) -> bool:        if payload_passphrase != settings.WEBHOOK_SECRET:            logger.warning("REJECTED: Passphrase Mismatch.")            return False
        if self.is_killswitch_active:            logger.error("REJECTED: Daily Killswitch Active.")            return False
        # Calculate daily drawdown        dd_pct = ((self.starting_daily_balance - self.current_equity) / self.starting_daily_balance) * 100.0        if dd_pct >= settings.MAX_DAILY_DRAWDOWN_PCT:            self.is_killswitch_active = True            logger.critical(f"KILLSWITCH ACTIVE: Daily Drawdown at {dd_pct:.2f}%. Halting execution.")            return False
        if not self.check_time_gate(phase):            logger.warning(f"REJECTED: Phase '{phase}' signal arrived outside authorized window.")            return False
        return True
